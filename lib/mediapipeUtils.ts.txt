import { KEYPOINTS_CONFIG } from "@/config/modelConfig";
import type { Holistic, Options, Results } from "@mediapipe/holistic";
import type {
  drawConnectors as DrawConnectorsFn,
  drawLandmarks as DrawLandmarksFn,
} from "@mediapipe/drawing_utils";

const MEDIAPIPE_VERSION = "0.5.1675471629";
const DRAWING_UTILS_VERSION = "0.3.1675466124";

const HOLISTIC_SCRIPT_URL = `https://cdn.jsdelivr.net/npm/@mediapipe/holistic@${MEDIAPIPE_VERSION}/holistic.js`;
const DRAWING_UTILS_SCRIPT_URL = `https://cdn.jsdelivr.net/npm/@mediapipe/drawing_utils@${DRAWING_UTILS_VERSION}/drawing_utils.js`;

type HolisticConstructor = typeof Holistic;
type DrawConnectors = typeof DrawConnectorsFn;
type DrawLandmarks = typeof DrawLandmarksFn;

interface MediaPipeGlobals {
  Holistic?: HolisticConstructor;
  POSE_CONNECTIONS?: [number, number][];
  HAND_CONNECTIONS?: [number, number][];
  drawConnectors?: DrawConnectors;
  drawLandmarks?: DrawLandmarks;
}

let holistic: Holistic | null = null;
let resolveDetection: ((results: Results) => void) | null = null;

function getGlobals(): MediaPipeGlobals {
  return (typeof window !== "undefined" ? window : {}) as MediaPipeGlobals;
}

function loadScript(src: string): Promise<void> {
  return new Promise((resolve, reject) => {
    const existing = document.querySelector<HTMLScriptElement>(
      `script[src="${src}"]`,
    );
    if (existing) {
      resolve();
      return;
    }

    const script = document.createElement("script");
    script.src = src;
    script.async = true;
    script.onload = () => resolve();
    script.onerror = () =>
      reject(new Error(`Gagal memuat script MediaPipe: ${src}`));
    document.body.appendChild(script);
  });
}

export async function initializeMediaPipe(): Promise<boolean> {
  if (typeof window === "undefined") return false;

  const globals = getGlobals();
  if (globals.Holistic && globals.drawConnectors) {
    return true;
  }

try {
      await Promise.all([
        loadScript(DRAWING_UTILS_SCRIPT_URL),
        loadScript(HOLISTIC_SCRIPT_URL),
      ]);

      holistic = new (getGlobals().Holistic as HolisticConstructor)({
      locateFile: (file: string) =>
        `https://cdn.jsdelivr.net/npm/@mediapipe/holistic@${MEDIAPIPE_VERSION}/${file}`,
    });

    const options: Options = {
      minDetectionConfidence: 0.5,
      minTrackingConfidence: 0.5,
      refineFaceLandmarks: false,
      enableSegmentation: false,
      modelComplexity: 1,
    };
    holistic.setOptions(options);

    holistic.onResults((results) => {
      if (resolveDetection) {
        resolveDetection(results);
        resolveDetection = null;
      }
    });

    await holistic.initialize();

    console.log("✅ MediaPipe Holistic siap");
    return true;
  } catch (error) {
    console.error("❌ Gagal inisialisasi MediaPipe:", error);
    return false;
  }
}

export async function detectPoseLandmarks(
  video: HTMLVideoElement | HTMLCanvasElement,
): Promise<Results | null> {
  if (!holistic) return null;

  try {
    return new Promise((resolve) => {
      resolveDetection = resolve;

      holistic!.send({ image: video }).catch(() => {
        if (resolveDetection === resolve) {
          resolveDetection = null;
          resolve(null);
        }
      });

      setTimeout(() => {
        if (resolveDetection === resolve) {
          resolveDetection = null;
          resolve(null);
        }
      }, 1000);
    });
  } catch (error) {
    console.error("Error mendeteksi landmark:", error);
    return null;
  }
}

export function extractKeypoints(detectionResult: Results): Float32Array {
  const keypoints = new Float32Array(KEYPOINTS_CONFIG.TOTAL_FEATURES);
  let featureIndex = 0;

  const writePose = () => {
    if (detectionResult.poseLandmarks) {
      for (let i = 0; i < KEYPOINTS_CONFIG.POSE_KEYPOINTS; i++) {
        const lm = detectionResult.poseLandmarks[i];
        keypoints[featureIndex++] = lm?.x ?? 0;
        keypoints[featureIndex++] = lm?.y ?? 0;
        keypoints[featureIndex++] = lm?.z ?? 0;
        keypoints[featureIndex++] = lm?.visibility ?? 0;
      }
    } else {
      featureIndex +=
        KEYPOINTS_CONFIG.POSE_KEYPOINTS * KEYPOINTS_CONFIG.POSE_DIMENSIONS;
    }
  };

  const writeHand = (landmarks: Results["leftHandLandmarks"]) => {
    if (landmarks) {
      for (let i = 0; i < KEYPOINTS_CONFIG.LEFT_HAND_KEYPOINTS; i++) {
        const lm = landmarks[i];
        keypoints[featureIndex++] = lm?.x ?? 0;
        keypoints[featureIndex++] = lm?.y ?? 0;
        keypoints[featureIndex++] = lm?.z ?? 0;
      }
    } else {
      featureIndex +=
        KEYPOINTS_CONFIG.LEFT_HAND_KEYPOINTS * KEYPOINTS_CONFIG.HAND_DIMENSIONS;
    }
  };

  try {
    writePose();
    writeHand(detectionResult.leftHandLandmarks);
    writeHand(detectionResult.rightHandLandmarks);
    return keypoints;
  } catch (error) {
    console.error("Error ekstrak keypoints:", error);
    return keypoints;
  }
}

export function drawLandmarks(
  canvas: HTMLCanvasElement,
  detectionResult: Results | null,
  showLandmarks: boolean,
) {
  try {
    const canvasCtx = canvas?.getContext("2d");
    if (!canvasCtx) return;

    canvasCtx.clearRect(0, 0, canvas.width, canvas.height);

    if (!showLandmarks || !detectionResult) return;

    const { POSE_CONNECTIONS, HAND_CONNECTIONS, drawConnectors } =
      getGlobals();
    if (!drawConnectors) return;

    canvasCtx.save();

    if (detectionResult.poseLandmarks && POSE_CONNECTIONS) {
      drawConnectors(canvasCtx, detectionResult.poseLandmarks, POSE_CONNECTIONS, {
        color: "rgb(20,184,166)",
        lineWidth: 1,
      });
      (getGlobals().drawLandmarks as DrawLandmarks)?.(
        canvasCtx,
        detectionResult.poseLandmarks,
        { color: "rgb(13,148,136)", radius: 1 },
      );
    }

    if (detectionResult.leftHandLandmarks && HAND_CONNECTIONS) {
      drawConnectors(canvasCtx, detectionResult.leftHandLandmarks, HAND_CONNECTIONS, {
        color: "rgb(34,211,238)",
        lineWidth: 2,
      });
      (getGlobals().drawLandmarks as DrawLandmarks)?.(
        canvasCtx,
        detectionResult.leftHandLandmarks,
        { color: "rgb(8,145,178)", radius: 4 },
      );
    }

    if (detectionResult.rightHandLandmarks && HAND_CONNECTIONS) {
      drawConnectors(canvasCtx, detectionResult.rightHandLandmarks, HAND_CONNECTIONS, {
        color: "rgb(251,191,36)",
        lineWidth: 2,
      });
      (getGlobals().drawLandmarks as DrawLandmarks)?.(
        canvasCtx,
        detectionResult.rightHandLandmarks,
        { color: "rgb(217,119,6)", radius: 4 },
      );
    }

    canvasCtx.restore();
  } catch (error) {
    console.error("Error menggambar landmark:", error);
  }
}

export function isHandDetected(detectionResult: Results | null): boolean {
  if (!detectionResult) return false;
  return (
    !!detectionResult.leftHandLandmarks || !!detectionResult.rightHandLandmarks
  );
}

export function isPoseDetected(detectionResult: Results | null): boolean {
  if (!detectionResult) return false;
  return !!detectionResult.poseLandmarks;
}

/**
 * Estimasi jarak wajah ke kamera berdasarkan lebar bahu.
 * Focal length standar webcam laptop ~ 700 px.
 */
export function calculateShoulderDistance(
  detectionResult: Results | null,
  imageWidth: number,
  imageHeight: number,
  focalLength: number = 700,
): number {
  if (!detectionResult || !detectionResult.poseLandmarks) return 0;

  const leftShoulder = detectionResult.poseLandmarks[11];
  const rightShoulder = detectionResult.poseLandmarks[12];

  if (!leftShoulder || !rightShoulder) return 0;

  const x1 = leftShoulder.x * imageWidth;
  const y1 = leftShoulder.y * imageHeight;
  const x2 = rightShoulder.x * imageWidth;
  const y2 = rightShoulder.y * imageHeight;

  const wPixel = Math.sqrt(Math.pow(x2 - x1, 2) + Math.pow(y2 - y1, 2));

  if (wPixel < 10) return 0;

  const W_BAHU = 40; // rata-rata lebar bahu 40cm
  return Math.round((W_BAHU * focalLength) / wPixel);
}
