"use client";

import Link from "next/link";
import { usePathname } from "next/navigation";
import { useState } from "react";
import {
  Hand,
  Home,
  BookOpen,
  Settings,
  Sparkles,
  X,
  Menu,
  GraduationCap,
} from "lucide-react";
import { ThemeToggle } from "@/components/theme-toggle";
import { cn } from "@/lib/utils";

const NAV_ITEMS = [
  { href: "/", label: "Beranda", icon: Home },
  { href: "/kosakata", label: "Kosakata", icon: BookOpen },
  { href: "/latihan", label: "Latihan", icon: GraduationCap },
];

function Brand() {
  return (
    <Link href="/" className="flex items-center gap-3 px-2">
      <span className="grid size-10 shrink-0 place-items-center rounded-2xl bg-gradient-to-br from-teal-500 to-emerald-600 text-white shadow-float">
        <Hand className="size-5" />
      </span>
      <span className="leading-tight">
        <span className="block text-base font-extrabold tracking-tight text-foreground">
          SIBI Studio
        </span>
        <span className="block text-[11px] font-medium text-muted-foreground">
          Pengenalan Bahasa Isyarat
        </span>
      </span>
    </Link>
  );
}

function NavLink({
  href,
  label,
  icon: Icon,
  onNavigate,
}: {
  href: string;
  label: string;
  icon: typeof Home;
  onNavigate?: () => void;
}) {
  const pathname = usePathname();
  const active =
    href === "/" ? pathname === "/" : pathname.startsWith(href);

  return (
    <Link
      href={href}
      onClick={onNavigate}
      className={cn(
        "flex items-center gap-3 rounded-2xl px-3.5 py-2.5 text-sm font-semibold transition-colors",
        active
          ? "bg-primary text-primary-foreground shadow-panel"
          : "text-muted-foreground hover:bg-surface-2 hover:text-foreground",
      )}
    >
      <Icon className="size-5" />
      {label}
    </Link>
  );
}

export function Sidebar() {
  const [open, setOpen] = useState(false);

  return (
    <>
      <header className="fixed inset-x-0 top-0 z-40 flex h-16 items-center justify-between border-b border-border bg-background/85 px-4 backdrop-blur lg:hidden">
        <Brand />
        <button
          onClick={() => setOpen(true)}
          aria-label="Buka menu"
          className="grid size-10 place-items-center rounded-2xl border border-border bg-surface-2 text-foreground"
        >
          <Menu className="size-5" />
        </button>
      </header>

      {open && (
        <div
          className="fixed inset-0 z-50 bg-black/50 backdrop-blur-sm lg:hidden"
          onClick={() => setOpen(false)}
        />
      )}

      <aside
        className={cn(
          "fixed inset-y-0 left-0 z-50 flex w-72 flex-col border-r border-border bg-background px-4 py-6 transition-transform duration-300 lg:translate-x-0",
          open ? "translate-x-0" : "-translate-x-full",
        )}
      >
        <div className="flex items-center justify-between">
          <Brand />
          <button
            onClick={() => setOpen(false)}
            aria-label="Tutup menu"
            className="grid size-9 place-items-center rounded-xl border border-border text-muted-foreground lg:hidden"
          >
            <X className="size-4" />
          </button>
        </div>

        <nav className="mt-8 flex flex-col gap-1">
          {NAV_ITEMS.map((item) => (
            <NavLink
              key={item.href}
              {...item}
              onNavigate={() => setOpen(false)}
            />
          ))}
        </nav>

        <div className="mt-auto flex flex-col gap-3">
          <div className="rounded-2xl border border-border bg-gradient-to-br from-teal-500/10 to-emerald-500/10 p-4">
            <div className="flex items-center gap-2 text-sm font-bold text-foreground">
              <Sparkles className="size-4 text-primary" />
              Tips Praktik
            </div>
            <p className="mt-1.5 text-xs leading-relaxed text-muted-foreground">
              Posisikan kedua tangan terlihat penuh di layar dan jaga jarak ±1
              meter dari kamera agar deteksi lebih akurat.
            </p>
          </div>

          <div className="flex items-center justify-between border-t border-border pt-4">
            <Link
              href="/pengaturan"
              className="flex items-center gap-3 rounded-xl px-2 py-2 text-sm font-semibold text-muted-foreground transition-colors hover:text-foreground"
            >
              <Settings className="size-5" />
              Pengaturan
            </Link>
            <ThemeToggle />
          </div>
        </div>
      </aside>
    </>
  );
}
